async function handleLogin() { const login = document.getElementById('login-user').value.trim(); const pass = document.getElementById('login-pass').value.trim(); if(!login || !pass) { alert('Заполните все поля!'); return; } // Формируем email так же, как при регистрации const email = `${login}@coziswallet.app`; try { // 1. Входим через Firebase Auth [citation:2] const userCredential = await auth.signInWithEmailAndPassword(email, pass); const user = userCredential.user; // 2. Загружаем профиль из Firestore по UID const userDoc = await db.collection('users').doc(user.uid).get(); if (!userDoc.exists) { alert('Ошибка: профиль пользователя не найден.'); await auth.signOut(); return; } const userData = userDoc.data(); // 3. Восстанавливаем состояние приложения currentAuthUserDocId = user.uid; currentAuthUser = { id: user.uid, ...userData }; appState.cards = userData.cards || []; appState.pin = userData.pin || ''; appState.walletId = userData.walletId || generateWalletId(); if (!userData.walletId) { await syncUserDataToCloud(); } // 4. Переход в зависимости от наличия PIN-кода if(!appState.pin) { goToScreen('screen-pin-setup'); } else { appState.modePin = 'lock'; resetPinInput(); goToScreen('screen-pin-lock'); } } catch (e) { console.error(e); // Обработка ошибок Firebase Auth (например, неверный пароль) if (e.code === 'auth/wrong-password' || e.code === 'auth/user-not-found') { alert('Неверный логин или пароль.'); } else { alert('Ошибка соединения с облачной базой данных: ' + e.message); } } }
